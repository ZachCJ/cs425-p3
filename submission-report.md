## Design

### 1. /0 (Default Route / Entire IPv4 Address Space)

- Netmask Avoidance of Undefined Behavior (cidr_mask):
  - In C, bit-shifting a 32-bit integer by 32 bits (i.e. 0xFFFFFFFFU << (32 - 0)) is Undefined Behavior (UB) per C standard 6.5.7.
  - Handled with an explicit zero guard:

        if (prefix == 0) {
          return 0; // 0.0.0.0
        }

- Network & Broadcast Addresses (cidr_network, cidr_broadcast):
  - Network address: addr & 0 = 0.0.0.0.
  - Broadcast address: addr | ~0 = 255.255.255.255.
- 64-bit Host Count (cidr_host_count):
  - The standard host formula is 2^{32-prefix} - 2. For /0, 2³² = 4, 294, 967, 296, which overflows a standard 32-bit uint32_t
      (UINT32_MAX is 4, 294, 967, 295).
  - Evaluated using 64-bit arithmetic:

        return (1ULL << (32 - block.prefix)) - 2; // Returns 4,294,967,294

- First & Last Host (cidr_first_host, cidr_last_host):
  - First host: 0.0.0.1 (network + 1).
  - Last host: 255.255.255.254 (broadcast - 1).
- Routing & Containment (cidr_contains, table_lookup):
  - Evaluates (addr & 0) == (0 & 0), matching all IPv4 addresses. Because its prefix length is 0, any route with a prefix > 0 will win over it during longest prefix matching.

### 2. /31 (Point-to-Point Links — RFC 3021)

- Netmask:
  - cidr_mask(31) returns 0xFFFFFFFE (255.255.255.254).
- RFC 3021 Special Host Semantics:
  - Traditional subnetting reserves 2 addresses (network and broadcast), which would leave 2 - 2 = 0 hosts. RFC 3021 defines point-
      to-point links where both addresses are assigned to the two router interfaces:
  - First Host: The network address itself (net).
  - Last Host: The broadcast address itself (bcast).

        uint32_t cidr_first_host(cidr_t block) {
          uint32_t net = cidr_network(block);
          if (block.prefix >= 31) {
            return net;
          }
          return net + 1;
        }

        uint32_t cidr_last_host(cidr_t block) {
          uint32_t bcast = cidr_broadcast(block);
          if (block.prefix >= 31) {
            return bcast;
          }
          return bcast - 1;
        }

  - Host Count: Exactly 2 hosts:

        if (block.prefix == 31) {
          return 2;
        }

### 3. /32 (Single Host / Loopback)

- Netmask:
  - cidr_mask(32) returns 0xFFFFFFFF (255.255.255.255). Any prefix > 32 is clamped to 32.
- Single-Host Semantics:
  - Network and broadcast addresses are identical to the host address itself.
  - First Host & Last Host: Equal to the address itself (shares the prefix >= 31 check with /31).
  -1 Host Count: Exactly 1 host:

        if (block.prefix >= 32) {
          return 1;
        }

- Lookup:
  - Matches only the exact destination IP. In table_lookup, prefix 32 is the maximum possible prefix length and always wins when matched.

/* ------------------------------------------------------------------------
 * Task 1 - Address parsing and formatting
 * --------------------------------------------------------------------- */

/**
 * @brief Helper to parse a single decimal octet from a substring [start, end).
 *
 * An octet must be 1 to 3 decimal digits, representing a value from 0 to 255.
 * Leading zeros are rejected (e.g. "01", "00", "092"), though a single "0" is valid.
 *
 * @param start Pointer to start of octet.
 * @param end Pointer to character immediately following octet.
 * @param octet Where to store the parsed octet value.
 * @return true if valid octet, false otherwise.
 */
static bool parse_octet_slice(const char *start, const char *end, uint8_t *octet) {
  if (!start || !end || end <= start) {
    return false;
  }
  size_t len = (size_t)(end - start);
  if (len > 3) {
    return false;
  }
  if (len > 1 && start[0] == '0') {
    return false;
  }
  unsigned int val = 0;
  for (size_t i = 0; i < len; i++) {
    if (start[i] < '0' || start[i] > '9') {
      return false;
    }
    val = val * 10 + (unsigned int)(start[i] - '0');
  }
  if (val > 255) {
    return false;
  }
  *octet = (uint8_t)val;
  return true;
}

/**
 * @brief Parse a dotted quad such as "192.168.1.10".
 *
 * @param text The string to parse.
 * @param addr Where the address is stored on success; untouched on failure.
 * @return true if text is a valid dotted quad, false otherwise (including NULL).
 */
bool ip_parse(const char *text, uint32_t *addr) {
  if (!text || !addr) {
    return false;
  }

  uint32_t result = 0;
  const char *p = text;

  for (int i = 0; i < 4; i++) {
    const char *start = p;
    const char *end = NULL;

    if (i < 3) {
      end = strchr(p, '.');
      if (!end) {
        return false;
      }
      p = end + 1;
    } else {
      end = start + strlen(start);
    }

    uint8_t octet = 0;
    if (!parse_octet_slice(start, end, &octet)) {
      return false;
    }

    result = (result << 8) | octet;
  }

  *addr = result;
  return true;
}

/**
 * @brief Format an address as a dotted quad.
 * @param addr The address, host byte order.
 * @param out A buffer of at least IP_STR_LEN bytes.
 */
void ip_format(uint32_t addr, char out[IP_STR_LEN]) {
  if (!out) {
    return;
  }
  snprintf(out, IP_STR_LEN, "%u.%u.%u.%u",
           (addr >> 24) & 0xFF,
           (addr >> 16) & 0xFF,
           (addr >> 8) & 0xFF,
           addr & 0xFF);
}

/**
 * @brief Parse a block in CIDR notation such as "192.168.1.0/24".
 *
 * @param text The string to parse.
 * @param block Where the block is stored on success; untouched on failure.
 * @return true if text is valid CIDR notation, false otherwise (including NULL).
 */
bool cidr_parse(const char *text, cidr_t *block) {
  if (!text || !block) {
    return false;
  }

  const char *slash = strchr(text, '/');
  if (!slash) {
    return false;
  }

  // Parse prefix after '/'
  const char *prefix_str = slash + 1;
  if (*prefix_str == '\0') {
    return false;
  }

  size_t plen = strlen(prefix_str);
  if (plen > 1 && prefix_str[0] == '0') {
    return false; // Leading zero (e.g. /024)
  }
  unsigned int prefix_val = 0;
  for (size_t i = 0; i < plen; i++) {
    if (prefix_str[i] < '0' || prefix_str[i] > '9') {
      return false;
    }
    prefix_val = prefix_val * 10 + (unsigned int)(prefix_str[i] - '0');
  }
  if (prefix_val > 32) {
    return false;
  }

  // Parse IP part before '/'
  uint32_t result = 0;
  const char *p = text;

  for (int i = 0; i < 4; i++) {
    const char *start = p;
    const char *end = NULL;

    if (i < 3) {
      end = strchr(p, '.');
      if (!end || end >= slash) {
        return false;
      }
      p = end + 1;
    } else {
      end = slash;
    }

    uint8_t octet = 0;
    if (!parse_octet_slice(start, end, &octet)) {
      return false;
    }

    result = (result << 8) | octet;
  }

  block->addr = result;
  block->prefix = (uint8_t)prefix_val;
  return true;
}

/**
 * @brief Format a block as "a.b.c.d/p", using the address exactly as stored.
 *
 * @param block The block.
 * @param out A buffer of at least CIDR_STR_LEN bytes.
 */
void cidr_format(cidr_t block, char out[CIDR_STR_LEN]) {
  if (!out) {
    return;
  }
  uint8_t prefix = block.prefix > 32 ? 32 : block.prefix;
  snprintf(out, CIDR_STR_LEN, "%u.%u.%u.%u/%u",
           (block.addr >> 24) & 0xFF,
           (block.addr >> 16) & 0xFF,
           (block.addr >> 8) & 0xFF,
           block.addr & 0xFF,
           prefix);
}

/* ------------------------------------------------------------------------
 * Task 2 - Subnet arithmetic
 * --------------------------------------------------------------------- */

/**
 * @brief The subnet mask for a prefix length: /24 is 255.255.255.0.
 *
 * @param prefix 0 to 32; anything larger is treated as 32.
 * @return The mask, host byte order.
 */
uint32_t cidr_mask(uint8_t prefix) {
  if (prefix == 0) {
    return 0;
  }
  if (prefix >= 32) {
    return 0xFFFFFFFFU;
  }
  return 0xFFFFFFFFU << (32 - prefix);
}

/** @brief The lowest address in the block: the address with the host bits cleared. */
uint32_t cidr_network(cidr_t block) {
  return block.addr & cidr_mask(block.prefix);
}

/** @brief The highest address in the block: the address with the host bits set. */
uint32_t cidr_broadcast(cidr_t block) {
  return block.addr | ~cidr_mask(block.prefix);
}

/**
 * @brief The first address that can be given to a host.
 *
 */
uint32_t cidr_first_host(cidr_t block) {
  uint32_t net = cidr_network(block);
  if (block.prefix >= 31) {
    return net;
  }
  return net + 1;
}

/**
 * @brief The last address that can be given to a host.
 *
 */
uint32_t cidr_last_host(cidr_t block) {
  uint32_t bcast = cidr_broadcast(block);
  if (block.prefix >= 31) {
    return bcast;
  }
  return bcast - 1;
}

/**
 * @brief How many addresses in the block can be given to hosts.
 *
 */
uint64_t cidr_host_count(cidr_t block) {
  if (block.prefix >= 32) {
    return 1;
  }
  if (block.prefix == 31) {
    return 2;
  }
  return (1ULL << (32 - block.prefix)) - 2;
}

/** @brief true if addr falls inside the block. */
bool cidr_contains(cidr_t block, uint32_t addr) {
  uint32_t mask = cidr_mask(block.prefix);
  return (addr & mask) == (block.addr & mask);
}

/** @brief true if the block's host bits are all zero, so it names a network. */
bool cidr_is_network(cidr_t block) {
  uint32_t mask = cidr_mask(block.prefix);
  return (block.addr & ~mask) == 0;
}

/**
 * @brief The i-th of n equally sized subnets of a block.
 *
 * @param block The block to split; it must be a network (cidr_is_network()).
 * @param n How many subnets, a power of two from 1 up.
 * @param i Which subnet, 0 to n - 1.
 * @param out Where the subnet is stored on success; untouched on failure.
 * @return false if out is NULL, the block is not a network, n is not a power
 *         of two, the split would need a prefix longer than 32, or i is out of
 *         range.
 */
bool cidr_subnet(cidr_t block, uint32_t n, uint32_t i, cidr_t *out) {
  if (!out || !cidr_is_network(block) || n == 0) {
    return false;
  }

  // n must be a power of two
  if ((n & (n - 1)) != 0) {
    return false;
  }

  // i must be in range 0 to n - 1
  if (i >= n) {
    return false;
  }

  // Determine number of borrowed bits k = log2(n)
  uint32_t temp = n;
  uint8_t k = 0;
  while (temp > 1) {
    temp >>= 1;
    k++;
  }

  uint32_t new_prefix = (uint32_t)block.prefix + k;
  if (new_prefix > 32) {
    return false;
  }

  uint32_t addr = block.addr;
  if (new_prefix > 0) {
    addr += (i << (32 - new_prefix));
  }

  out->addr = addr;
  out->prefix = (uint8_t)new_prefix;
  return true;
}

/* ------------------------------------------------------------------------
 * Task 3 - Longest prefix match
 * --------------------------------------------------------------------- */

/** Internal structure for the forwarding table. */
struct fwd_table {
  route_t *routes;
  size_t count;
  size_t capacity;
};

/**
 * @brief Parse one line of a forwarding table file.
 *
 * @param line The line.
 * @param route Where the route is stored on success; untouched on failure.
 * @return true if the line is a valid route, false otherwise (including NULL).
 */
bool route_parse(const char *line, route_t *route) {
  if (!line || !route) {
    return false;
  }

  const char *p = line;

  // 1. Skip leading whitespace
  while (*p && (*p == ' ' || *p == '\t' || *p == '\r' || *p == '\n')) {
    p++;
  }
  if (*p == '\0') {
    return false;
  }

  // 2. Field 1: Destination
  const char *f1_start = p;
  while (*p && *p != ' ' && *p != '\t' && *p != '\r' && *p != '\n') {
    p++;
  }
  size_t f1_len = (size_t)(p - f1_start);

  // Skip whitespace between field 1 and field 2
  while (*p && (*p == ' ' || *p == '\t' || *p == '\r' || *p == '\n')) {
    p++;
  }
  if (*p == '\0') {
    return false;
  }

  // 3. Field 2: Gateway
  const char *f2_start = p;
  while (*p && *p != ' ' && *p != '\t' && *p != '\r' && *p != '\n') {
    p++;
  }
  size_t f2_len = (size_t)(p - f2_start);

  // Skip whitespace between field 2 and field 3
  while (*p && (*p == ' ' || *p == '\t' || *p == '\r' || *p == '\n')) {
    p++;
  }
  if (*p == '\0') {
    return false;
  }

  // 4. Field 3: Device
  const char *f3_start = p;
  while (*p && *p != ' ' && *p != '\t' && *p != '\r' && *p != '\n') {
    p++;
  }
  size_t f3_len = (size_t)(p - f3_start);

  // Skip trailing whitespace
  while (*p && (*p == ' ' || *p == '\t' || *p == '\r' || *p == '\n')) {
    p++;
  }
  if (*p != '\0') {
    return false;
  }

  // Length sanity checks
  char token1[64];
  char token2[64];
  char token3[64];

  if (f1_len >= sizeof(token1) || f2_len >= sizeof(token2) || f3_len >= sizeof(token3)) {
    return false;
  }

  memcpy(token1, f1_start, f1_len);
  token1[f1_len] = '\0';

  memcpy(token2, f2_start, f2_len);
  token2[f2_len] = '\0';

  memcpy(token3, f3_start, f3_len);
  token3[f3_len] = '\0';

  // Parse destination: either "default" (0.0.0.0/0) or CIDR block with no host bits set
  cidr_t dest;
  if (strcmp(token1, "default") == 0) {
    dest.addr = 0;
    dest.prefix = 0;
  } else {
    if (!cidr_parse(token1, &dest)) {
      return false;
    }
    if (!cidr_is_network(dest)) {
      return false;
    }
  }

  // Parse gateway: either "direct" (0) or dotted quad IP
  uint32_t gateway = 0;
  if (strcmp(token2, "direct") == 0) {
    gateway = 0;
  } else {
    if (!ip_parse(token2, &gateway)) {
      return false;
    }
  }

  // Device must be 1 to DEV_NAME_LEN - 1 characters
  if (f3_len == 0 || f3_len >= DEV_NAME_LEN) {
    return false;
  }

  // Success: write to route
  route->dest = dest;
  route->gateway = gateway;
  memset(route->dev, 0, sizeof(route->dev));
  memcpy(route->dev, token3, f3_len);
  route->dev[f3_len] = '\0';

  return true;
}

/**
 * @brief Create an empty forwarding table.
 * @return The table, which the caller frees with table_destroy(), or NULL if
 *         memory could not be allocated.
 */
fwd_table_t *table_create(void) {
  fwd_table_t *table = (fwd_table_t *)malloc(sizeof(fwd_table_t));
  if (!table) {
    // GCOVR_EXCL_START
    return NULL;
  } // GCOVR_EXCL_STOP
  table->routes = NULL;
  table->count = 0;
  table->capacity = 0;
  return table;
}

/** @brief Free a table and everything in it. NULL is allowed and does nothing. */
void table_destroy(fwd_table_t *table) {
  if (!table) {
    return;
  }
  free(table->routes);
  free(table);
}

/** @brief The number of routes in the table; 0 for NULL. */
size_t table_count(const fwd_table_t *table) {
  if (!table) {
    return 0;
  }
  return table->count;
}

/**
 * @brief Add a copy of a route to the table, growing it as needed.
 * @return false if either argument is NULL, the destination has host bits
 *         set, the table already holds a route to the same block, or memory
 *         could not be allocated. The table is unchanged on failure.
 */
bool table_add(fwd_table_t *table, const route_t *route) {
  if (!table || !route) {
    return false;
  }

  // Destination cannot have host bits set, and prefix cannot exceed 32
  if (route->dest.prefix > 32 || !cidr_is_network(route->dest)) {
    return false;
  }

  // Table cannot already hold a route to the same block
  for (size_t idx = 0; idx < table->count; idx++) {
    if (table->routes[idx].dest.addr == route->dest.addr &&
        table->routes[idx].dest.prefix == route->dest.prefix) {
      return false;
    }
  }

  // Grow array if capacity reached
  if (table->count >= table->capacity) {
    size_t new_cap = table->capacity == 0 ? 8 : table->capacity * 2;
    route_t *new_routes = (route_t *)realloc(table->routes, new_cap * sizeof(route_t));
    if (!new_routes) {
      // GCOVR_EXCL_START
      return false;
    } // GCOVR_EXCL_STOP
    table->routes = new_routes;
    table->capacity = new_cap;
  }

  table->routes[table->count] = *route;
  table->count++;
  return true;
}

/**
 * @brief Helper to read a line of arbitrary length from a file stream.
 *
 * @param in Open file stream.
 * @return Heap-allocated string which caller must free, or NULL on EOF / error.
 */
static char *table_read_line(FILE *in) {
  size_t cap = 128;
  size_t len = 0;
  char *buf = (char *)malloc(cap);
  if (!buf) {
    // GCOVR_EXCL_START
    return NULL;
  } // GCOVR_EXCL_STOP

  int c = fgetc(in);
  if (c == EOF) {
    free(buf);
    return NULL;
  }

  while (c != EOF && c != '\n') {
    if (len + 1 >= cap) {
      size_t new_cap = cap * 2;
      char *new_buf = (char *)realloc(buf, new_cap);
      if (!new_buf) {
        // GCOVR_EXCL_START
        free(buf);
        return NULL;
      } // GCOVR_EXCL_STOP
      buf = new_buf;
      cap = new_cap;
    }
    buf[len++] = (char)c;
    c = fgetc(in);
  }
  buf[len] = '\0';
  return buf;
}

/**
 * @brief Read a forwarding table file and add every route in it.
 *
 * @param table The table to add to.
 * @param in The open file to read.
 * @param bad_line If not NULL, set to the 1-based number of the first line
 *        that failed, or 0 if the arguments were NULL or the file could not
 *        be read.
 * @return true if every line loaded. On false, routes from the lines before
 *         the bad one stay in the table.
 */
bool table_load(fwd_table_t *table, FILE *in, size_t *bad_line) {
  if (!table || !in) {
    if (bad_line) {
      *bad_line = 0;
    }
    return false;
  }

  size_t line_num = 0;
  char *line = NULL;

  while ((line = table_read_line(in)) != NULL) {
    line_num++;

    // Everything from '#' to end of line is a comment
    char *comment = strchr(line, '#');
    if (comment) {
      *comment = '\0';
    }

    // Check if line is blank once comment is removed
    const char *p = line;
    while (*p && (*p == ' ' || *p == '\t' || *p == '\r' || *p == '\n')) {
      p++;
    }
    if (*p == '\0') {
      free(line);
      continue;
    }

    // Parse route
    route_t r;
    if (!route_parse(line, &r)) {
      free(line);
      if (bad_line) {
        *bad_line = line_num;
      }
      return false;
    }

    // Add route to table
    if (!table_add(table, &r)) {
      free(line);
      if (bad_line) {
        *bad_line = line_num;
      }
      return false;
    }

    free(line);
  }

  return true;
}

/**
 * @brief Longest prefix match: the route whose block contains dest and has the
 *        longest prefix.
 *
 * @return The matching route, which stays owned by the table, or NULL if no
 *         route matches (or table is NULL).
 */
const route_t *table_lookup(const fwd_table_t *table, uint32_t dest) {
  if (!table || table->count == 0) {
    return NULL;
  }

  const route_t *best_match = NULL;

  for (size_t idx = 0; idx < table->count; idx++) {
    if (cidr_contains(table->routes[idx].dest, dest)) {
      if (!best_match || table->routes[idx].dest.prefix > best_match->dest.prefix) {
        best_match = &table->routes[idx];
      }
    }
  }

  return best_match;
}

/**
 * @brief Handle the 'split' command.
 *
 * @param cidr_str Network address with prefix length (e.g. 192.168.1.0/24).
 * @param count_str Number of subnets to create (power of two).
 * @return 0 on success, 1 on failure.
 */
static int handle_split(const char *cidr_str, const char *count_str) {
  cidr_t block;
  if (!cidr_parse(cidr_str, &block)) {
    fprintf(stderr, "myapp: '%s' is not a valid network/prefix\n", cidr_str);
    return 1;
  }

  if (!cidr_is_network(block)) {
    fprintf(stderr, "myapp: %s is not a network (host bits set)\n", cidr_str);
    return 1;
  }

  if (!count_str || *count_str == '\0') {
    fprintf(stderr, "myapp: invalid count\n");
    return 1;
  }

  for (const char *p = count_str; *p; p++) {
    if (*p < '0' || *p > '9') {
      fprintf(stderr, "myapp: invalid count '%s'\n", count_str);
      return 1;
    }
  }

  char *endptr = NULL;
  unsigned long long count_ull = strtoull(count_str, &endptr, 10);
  if (!endptr || *endptr != '\0' || count_ull == 0 || count_ull > 0xFFFFFFFFULL) {
    fprintf(stderr, "myapp: count out of range\n");
    return 1;
  }

  uint32_t count = (uint32_t)count_ull;

  if ((count & (count - 1)) != 0) {

/**
 * @brief Handle the 'lookup' command.
 *
 * @param filename Forwarding table file path.
 * @param argc Total argument count.
 * @param argv Argument vector starting from argv[3] as target addresses.
 * @return 0 on success, 1 on malformed address, 2 on file error, 3 on unreachable route.
 */
static int handle_lookup(const char *filename, int argc, char *argv[]) {
  FILE *f = fopen(filename, "r");
  if (!f) {
    fprintf(stderr, "myapp: %s: cannot open file: %s\n", filename, strerror(errno));
    return 2;
  }

  fwd_table_t *table = table_create();
  if (!table) {
    fclose(f);
    fprintf(stderr, "myapp: failed to allocate forwarding table\n");
    return 1;
  }

  size_t bad_line = 0;
  if (!table_load(table, f, &bad_line)) {
    fclose(f);
    table_destroy(table);
    if (bad_line > 0) {
      fprintf(stderr, "myapp: %s:%zu: not a valid route\n", filename, bad_line);
    } else {
      fprintf(stderr, "myapp: %s: could not read table\n", filename);
    }
    return 2;
  }
  fclose(f);

  bool has_malformed = false;
  bool has_unreachable = false;

  for (int i = 3; i < argc; i++) {
    uint32_t addr = 0;
    if (!ip_parse(argv[i], &addr)) {
      fprintf(stderr, "myapp: %s: malformed address\n", argv[i]);
      has_malformed = true;
      continue;
    }

    const route_t *r = table_lookup(table, addr);
    if (!r) {
      printf("%s unreachable\n", argv[i]);
      has_unreachable = true;
    } else if (r->gateway == 0) {
      printf("%s dev %s\n", argv[i], r->dev);
    } else {
      char gw_buf[IP_STR_LEN];
      ip_format(r->gateway, gw_buf);
      printf("%s via %s dev %s\n", argv[i], gw_buf, r->dev);
    }
  }

  table_destroy(table);

  if (has_malformed) {
    return 1;
  }
  if (has_unreachable) {
    return 3;
  }
  return 0;
}

/**
 * @brief Application entry point.
 */
int main(int argc, char *argv[]) {
  if (argc <= 1) {
    print_usage(stdout);
    return 0;
  }

  if (strcmp(argv[1], "info") == 0) {
    if (argc != 3) {
      fprintf(stderr, "myapp: 'info' requires exactly one <address>/<prefix> argument\n");
      print_usage(stderr);
      return 1;
    }
    return handle_info(argv[2]);
  }

  if (strcmp(argv[1], "split") == 0) {
    if (argc != 4) {
      fprintf(stderr, "myapp: 'split' requires <network>/<prefix> and <count>\n");
      print_usage(stderr);
      return 1;
    }
    return handle_split(argv[2], argv[3]);
  }

  if (strcmp(argv[1], "lookup") == 0) {
    if (argc < 4) {
      fprintf(stderr, "myapp: 'lookup' requires <table-file> and at least one <address>\n");
      print_usage(stderr);
      return 1;
    }
    return handle_lookup(argv[2], argc, argv);
  }

  fprintf(stderr, "myapp: unknown command '%s'\n", argv[1]);
  print_usage(stderr);
  return 1;
}

  TEST_ASSERT_TRUE(cidr_contains(onyx, make_ip(132, 178, 227, 50)));   // host inside
  TEST_ASSERT_TRUE(cidr_contains(onyx, make_ip(132, 178, 227, 127)));  // broadcast
  TEST_ASSERT_FALSE(cidr_contains(onyx, make_ip(132, 178, 227, 128))); // outside (next subnet)
  TEST_ASSERT_FALSE(cidr_contains(onyx, make_ip(132, 178, 227, 200))); // outside
  TEST_ASSERT_FALSE(cidr_contains(onyx, make_ip(10, 0, 0, 1)));        // different network

  // /0 contains everything
  cidr_t zero = make_cidr(0, 0, 0, 0, 0);
  TEST_ASSERT_TRUE(cidr_contains(zero, 0));
  TEST_ASSERT_TRUE(cidr_contains(zero, make_ip(192, 168, 1, 1)));
  TEST_ASSERT_TRUE(cidr_contains(zero, 0xFFFFFFFF));

  // /32 contains only that exact address
  cidr_t single = make_cidr(192, 168, 1, 10, 32);
  TEST_ASSERT_TRUE(cidr_contains(single, make_ip(192, 168, 1, 10)));
  TEST_ASSERT_FALSE(cidr_contains(single, make_ip(192, 168, 1, 9)));
  TEST_ASSERT_FALSE(cidr_contains(single, make_ip(192, 168, 1, 11)));

  // Prefix > 32 treated as 32
  cidr_t over = make_cidr(192, 168, 1, 10, 35);
  TEST_ASSERT_TRUE(cidr_contains(over, make_ip(192, 168, 1, 10)));
  TEST_ASSERT_FALSE(cidr_contains(over, make_ip(192, 168, 1, 11)));
}

void cidr_is_network_functionality_test(void) {
  // Host bits all zero -> true
  TEST_ASSERT_TRUE(cidr_is_network(make_cidr(192, 168, 1, 0, 24)));
  TEST_ASSERT_TRUE(cidr_is_network(make_cidr(132, 178, 227, 0, 25)));
  TEST_ASSERT_TRUE(cidr_is_network(make_cidr(132, 178, 227, 128, 25)));
  TEST_ASSERT_TRUE(cidr_is_network(make_cidr(0, 0, 0, 0, 0)));
  TEST_ASSERT_TRUE(cidr_is_network(make_cidr(10, 0, 0, 0, 8)));

  // Host bits not all zero -> false
  TEST_ASSERT_FALSE(cidr_is_network(make_cidr(192, 168, 1, 1, 24)));
  TEST_ASSERT_FALSE(cidr_is_network(make_cidr(192, 168, 1, 5, 24)));
  TEST_ASSERT_FALSE(cidr_is_network(make_cidr(132, 178, 227, 11, 25)));
  TEST_ASSERT_FALSE(cidr_is_network(make_cidr(1, 2, 3, 4, 0))); // for /0, all 32 bits are host bits!

  // /31: even addresses have lowest bit 0 -> network; odd addresses have bit 1 -> host
  TEST_ASSERT_TRUE(cidr_is_network(make_cidr(192, 168, 1, 4, 31)));
  TEST_ASSERT_FALSE(cidr_is_network(make_cidr(192, 168, 1, 5, 31)));

  // /32: 0 host bits, host bits are all zero -> true
  TEST_ASSERT_TRUE(cidr_is_network(make_cidr(192, 168, 1, 1, 32)));

  // Prefix > 32 treated as 32
  TEST_ASSERT_TRUE(cidr_is_network(make_cidr(192, 168, 1, 1, 40)));
}

void cidr_subnet_functionality_test(void) {
  cidr_t out = {0, 0};

  // Valid splitting: 192.168.1.0/24 into 4 subnets (all /26)
  cidr_t base24 = make_cidr(192, 168, 1, 0, 24);

  TEST_ASSERT_TRUE(cidr_subnet(base24, 4, 0, &out));
  TEST_ASSERT_EQUAL_UINT32(make_ip(192, 168, 1, 0), out.addr);
  TEST_ASSERT_EQUAL_UINT8(26, out.prefix);

  TEST_ASSERT_TRUE(cidr_subnet(base24, 4, 1, &out));
  TEST_ASSERT_EQUAL_UINT32(make_ip(192, 168, 1, 64), out.addr);
  TEST_ASSERT_EQUAL_UINT8(26, out.prefix);

  TEST_ASSERT_TRUE(cidr_subnet(base24, 4, 2, &out));
  TEST_ASSERT_EQUAL_UINT32(make_ip(192, 168, 1, 128), out.addr);
  TEST_ASSERT_EQUAL_UINT8(26, out.prefix);

  TEST_ASSERT_TRUE(cidr_subnet(base24, 4, 3, &out));
  TEST_ASSERT_EQUAL_UINT32(make_ip(192, 168, 1, 192), out.addr);
  TEST_ASSERT_EQUAL_UINT8(26, out.prefix);

  // Splitting into 1 subnet (n = 1): returns the block unchanged
  TEST_ASSERT_TRUE(cidr_subnet(base24, 1, 0, &out));
  TEST_ASSERT_EQUAL_UINT32(base24.addr, out.addr);
  TEST_ASSERT_EQUAL_UINT8(base24.prefix, out.prefix);

  // Splitting /0 into 2 subnets: 0.0.0.0/1 and 128.0.0.0/1
  cidr_t base0 = make_cidr(0, 0, 0, 0, 0);
  TEST_ASSERT_TRUE(cidr_subnet(base0, 2, 0, &out));
  TEST_ASSERT_EQUAL_UINT32(make_ip(0, 0, 0, 0), out.addr);
  TEST_ASSERT_EQUAL_UINT8(1, out.prefix);

  TEST_ASSERT_TRUE(cidr_subnet(base0, 2, 1, &out));
  TEST_ASSERT_EQUAL_UINT32(make_ip(128, 0, 0, 0), out.addr);
  TEST_ASSERT_EQUAL_UINT8(1, out.prefix);

  // Failure cases: out must remain untouched on failure!
  cidr_t sentinel = make_cidr(99, 99, 99, 99, 99);
  out = sentinel;

  // 1. out is NULL
  TEST_ASSERT_FALSE(cidr_subnet(base24, 4, 0, NULL));

  // 2. Block is not a network (host bits set)
  cidr_t notNet = make_cidr(192, 168, 1, 5, 24);
  TEST_ASSERT_FALSE(cidr_subnet(notNet, 4, 0, &out));
  TEST_ASSERT_EQUAL_UINT32(sentinel.addr, out.addr);
  TEST_ASSERT_EQUAL_UINT8(sentinel.prefix, out.prefix);

  // 3. n is 0 or not a power of two
  TEST_ASSERT_FALSE(cidr_subnet(base24, 0, 0, &out));
  TEST_ASSERT_EQUAL_UINT32(sentinel.addr, out.addr);
  TEST_ASSERT_FALSE(cidr_subnet(base24, 3, 0, &out));
  TEST_ASSERT_EQUAL_UINT32(sentinel.addr, out.addr);
  TEST_ASSERT_FALSE(cidr_subnet(base24, 5, 0, &out));
  TEST_ASSERT_EQUAL_UINT32(sentinel.addr, out.addr);
  TEST_ASSERT_FALSE(cidr_subnet(base24, 6, 0, &out));
  TEST_ASSERT_EQUAL_UINT32(sentinel.addr, out.addr);

  // 4. i is out of range (i >= n)
  TEST_ASSERT_FALSE(cidr_subnet(base24, 4, 4, &out));
  TEST_ASSERT_EQUAL_UINT32(sentinel.addr, out.addr);
  TEST_ASSERT_FALSE(cidr_subnet(base24, 4, 5, &out));
  TEST_ASSERT_EQUAL_UINT32(sentinel.addr, out.addr);

  // 5. Split would need a prefix longer than 32
  // /32 cannot be split (even n = 2 would need /33)
  cidr_t base32 = make_cidr(192, 168, 1, 1, 32);
  TEST_ASSERT_FALSE(cidr_subnet(base32, 2, 0, &out));
  TEST_ASSERT_EQUAL_UINT32(sentinel.addr, out.addr);

  // /30 split into 8 subnets borrows 3 bits -> prefix 33 > 32
  cidr_t base30 = make_cidr(192, 168, 1, 0, 30);
  TEST_ASSERT_FALSE(cidr_subnet(base30, 8, 0, &out));
  TEST_ASSERT_EQUAL_UINT32(sentinel.addr, out.addr);
}

void route_parse_functionality_test(void) {
  route_t *r = (route_t *)malloc(sizeof(route_t));

  // Valid route: regular CIDR with dotted IP gateway
  TEST_ASSERT_TRUE(route_parse("10.0.0.0/8 10.1.0.1 eth1", r));
  TEST_ASSERT_EQUAL_UINT32(make_ip(10, 0, 0, 0), r->dest.addr);
  TEST_ASSERT_EQUAL_UINT8(8, r->dest.prefix);
  TEST_ASSERT_EQUAL_UINT32(make_ip(10, 1, 0, 1), r->gateway);
  TEST_ASSERT_EQUAL_STRING("eth1", r->dev);

  // Valid route: "default" destination -> 0.0.0.0/0
  TEST_ASSERT_TRUE(route_parse("default 132.178.227.1 eno1", r));
  TEST_ASSERT_EQUAL_UINT32(0, r->dest.addr);
  TEST_ASSERT_EQUAL_UINT8(0, r->dest.prefix);
  TEST_ASSERT_EQUAL_UINT32(make_ip(132, 178, 227, 1), r->gateway);
  TEST_ASSERT_EQUAL_STRING("eno1", r->dev);

  // Valid route: "direct" gateway -> 0
  TEST_ASSERT_TRUE(route_parse("132.178.227.0/25 direct eno1", r));
  TEST_ASSERT_EQUAL_UINT32(make_ip(132, 178, 227, 0), r->dest.addr);
  TEST_ASSERT_EQUAL_UINT8(25, r->dest.prefix);
  TEST_ASSERT_EQUAL_UINT32(0, r->gateway);
  TEST_ASSERT_EQUAL_STRING("eno1", r->dev);

  // Valid route: "default" with "direct"
  TEST_ASSERT_TRUE(route_parse("default direct eth0", r));
  TEST_ASSERT_EQUAL_UINT32(0, r->dest.addr);
  TEST_ASSERT_EQUAL_UINT8(0, r->dest.prefix);
  TEST_ASSERT_EQUAL_UINT32(0, r->gateway);
  TEST_ASSERT_EQUAL_STRING("eth0", r->dev);

  // Valid route: leading/trailing whitespace, multiple spaces and tabs, trailing newline
  TEST_ASSERT_TRUE(route_parse(" \t 192.168.0.0/16 \t\t 10.1.0.1 \t eth1 \r\n", r));
  TEST_ASSERT_EQUAL_UINT32(make_ip(192, 168, 0, 0), r->dest.addr);
  TEST_ASSERT_EQUAL_UINT8(16, r->dest.prefix);
  TEST_ASSERT_EQUAL_UINT32(make_ip(10, 1, 0, 1), r->gateway);
  TEST_ASSERT_EQUAL_STRING("eth1", r->dev);

  // Valid route: maximum device name length (DEV_NAME_LEN - 1 = 15 chars)
  TEST_ASSERT_TRUE(route_parse("10.0.0.0/8 direct 123456789012345", r));
  TEST_ASSERT_EQUAL_STRING("123456789012345", r->dev);

  // Failure cases: r must remain untouched on failure!
  route_t * sentinel = make_route(make_cidr(99, 99, 99, 99, 99), 0x12345678, "sentinel");
  free(r);
  r = sentinel;

  // 1. NULL arguments (Safety Check)
  TEST_ASSERT_FALSE(route_parse(NULL, r));
  TEST_ASSERT_EQUAL_UINT32(sentinel->dest.addr, r->dest.addr);
  TEST_ASSERT_FALSE(route_parse("default direct eth0", NULL));

  // 2. Destination has host bits set
  TEST_ASSERT_FALSE(route_parse("192.168.1.5/24 direct eth0", r));
  TEST_ASSERT_EQUAL_UINT32(sentinel->dest.addr, r->dest.addr);
  TEST_ASSERT_FALSE(route_parse("132.178.227.11/25 10.1.0.1 eth0", r));
  TEST_ASSERT_EQUAL_UINT32(sentinel->dest.addr, r->dest.addr);

  // 3. Malformed destination
  TEST_ASSERT_FALSE(route_parse("10.0.0/8 direct eth0", r));
  TEST_ASSERT_FALSE(route_parse("10.0.0.0/33 direct eth0", r));
  TEST_ASSERT_FALSE(route_parse("10.0.0.0 direct eth0", r));
  TEST_ASSERT_FALSE(route_parse("defaul direct eth0", r));

  // 4. Malformed gateway
  TEST_ASSERT_FALSE(route_parse("10.0.0.0/8 10.1.0 eth1", r));
  TEST_ASSERT_FALSE(route_parse("10.0.0.0/8 256.1.1.1 eth1", r));
  TEST_ASSERT_FALSE(route_parse("10.0.0.0/8 010.0.0.1 eth1", r));
  TEST_ASSERT_FALSE(route_parse("10.0.0.0/8 indirectly eth1", r));

  // 5. Invalid device: empty or too long (>= DEV_NAME_LEN chars)
  TEST_ASSERT_FALSE(route_parse("10.0.0.0/8", r));
  TEST_ASSERT_FALSE(route_parse("10.0.0.0/8 direct", r));
  TEST_ASSERT_FALSE(route_parse("10.0.0.0/8 direct 1234567890123456", r));
  // Overload token
  TEST_ASSERT_FALSE(route_parse("10.0.0.0/8 direct 123456788912345678912345678912345678912345678912345678912345678912", r));

  // 6. Extra fields / comments not stripped by route_parse
  TEST_ASSERT_FALSE(route_parse("default direct eth0 extra", r));
  TEST_ASSERT_FALSE(route_parse("default direct eth0 # comment", r));
  TEST_ASSERT_FALSE(route_parse("", r));



  free(sentinel);
}

void table_add_functionality_test(void) {
  fwd_table_t *table = table_create();
  TEST_ASSERT_NOT_NULL(table);

  // Add valid route
  route_t * r1 = make_route(make_cidr(10, 0, 0, 0, 8), make_ip(10, 1, 0, 1), "eth1");
  TEST_ASSERT_TRUE(table_add(table, r1));
  TEST_ASSERT_EQUAL_size_t(1, table_count(table));

  // Add another route
  route_t * r2 = make_route(make_cidr(10, 1, 0, 0, 16), 0, "eth1");
  TEST_ASSERT_TRUE(table_add(table, r2));
  TEST_ASSERT_EQUAL_size_t(2, table_count(table));

  // Add default route
  route_t * rdef = make_route(make_cidr(0, 0, 0, 0, 0), make_ip(132, 178, 227, 1), "eno1");
  TEST_ASSERT_TRUE(table_add(table, rdef));
  TEST_ASSERT_EQUAL_size_t(3, table_count(table));

  // Failure: duplicate destination block (even with different gateway or device)
  route_t * r_dup = make_route(make_cidr(10, 0, 0, 0, 8), make_ip(10, 2, 0, 1), "eth2");
  TEST_ASSERT_FALSE(table_add(table, r_dup));
  TEST_ASSERT_EQUAL_size_t(3, table_count(table));

  // Failure: duplicate default route
  route_t * rdef_dup = make_route(make_cidr(0, 0, 0, 0, 0), 0, "eth0");
  TEST_ASSERT_FALSE(table_add(table, rdef_dup));
  TEST_ASSERT_EQUAL_size_t(3, table_count(table));

  // Failure: destination has host bits set
  route_t * r_hostbits = make_route(make_cidr(192, 168, 1, 5, 24), 0, "eth0");
  TEST_ASSERT_FALSE(table_add(table, r_hostbits));
  TEST_ASSERT_EQUAL_size_t(3, table_count(table));

  // Failure: NULL arguments
  TEST_ASSERT_FALSE(table_add(NULL, r1));
  TEST_ASSERT_FALSE(table_add(table, NULL));
  TEST_ASSERT_EQUAL_size_t(3, table_count(table));

  table_destroy(table);
  free(r1);
  free(r2);
  free(rdef);
  free(r_dup);
  free(rdef_dup);
  free(r_hostbits);
}

void table_load_functionality_test(void) {

  const char *valid_content =

    "default            132.178.227.1    eno1\n"

    "10.0.0.0/8         10.1.0.1         eth1\n"

    "192.168.0.0/16     10.1.0.1         eth1\n";

  FILE *f = create_temp_file(valid_content);

  TEST_ASSERT_TRUE(table_load(table, f, &bad_line));
  TEST_ASSERT_EQUAL_size_t(7, table_count(table));
  fclose(f);
  table_destroy(table);

    "default 132.178.227.1 eno1\n"

  TEST_ASSERT_EQUAL_size_t(2, table_count(long_table));
  fclose(long_f);

  bad_line = 0;
  TEST_ASSERT_FALSE(table_load(table, f, &bad_line));
  TEST_ASSERT_EQUAL_size_t(4, bad_line);
  TEST_ASSERT_EQUAL_size_t(1, table_count(table));
  fclose(f);
  table_destroy(table);

  // Duplicate route with NULL bad_line
  table = table_create();
  f = create_temp_file(dup_content);
  TEST_ASSERT_NOT_NULL(f);
  TEST_ASSERT_FALSE(table_load(table, f, NULL));
  fclose(f);
  table_destroy(table);

  // File with oneline comment
  const char *content =

    "default 132.178.227.1 eno1\n"
    "10.0.0.0/8 direct eth1\n"
    "# Should be no issue\n"
    "10.1.0.0/16 direct eth1\n";

  TEST_ASSERT_NOT_NULL(f);
  bad_line = 0;
  TEST_ASSERT_TRUE(table_load(table, f, &bad_line));

  table_destroy(table);

void table_lookup_functionality_test(void) {
  // NULL table returns NULL
  TEST_ASSERT_NULL(table_lookup(NULL, make_ip(10, 1, 2, 5)));

  // Empty table returns NULL
  fwd_table_t *table = table_create();
  TEST_ASSERT_NULL(table_lookup(table, make_ip(10, 1, 2, 5)));

  // Table without default route: unmatched address returns NULL
  route_t * r_ten = make_route(make_cidr(10, 0, 0, 0, 8), make_ip(10, 1, 0, 1), "eth1");
  table_add(table, r_ten);
  TEST_ASSERT_NULL(table_lookup(table, make_ip(192, 168, 1, 1)));

  // Match inside single route
  const route_t *match = table_lookup(table, make_ip(10, 5, 5, 5));
  TEST_ASSERT_NOT_NULL(match);
  TEST_ASSERT_EQUAL_UINT32(make_ip(10, 0, 0, 0), match->dest.addr);
  TEST_ASSERT_EQUAL_STRING("eth1", match->dev);
  table_destroy(table);
  free(r_ten);

  // Longest Prefix Match test (Canvas Assignment Onyx example)
  // Routes:
  // default            132.178.227.1    eno1
  // 132.178.227.0/25   direct           eno1
  // 10.0.0.0/8         10.1.0.1         eth1
  // 10.1.0.0/16        direct           eth1
  // 10.1.2.0/24        direct           eth2
  // 10.1.2.128/25      10.1.2.1         eth2
  // 192.168.0.0/16     10.1.0.1         eth1
  table = table_create();
  route_t * routes[7] = {
    make_route(make_cidr(0, 0, 0, 0, 0), make_ip(132, 178, 227, 1), "eno1"),
    make_route(make_cidr(132, 178, 227, 0, 25), 0, "eno1"),
    make_route(make_cidr(10, 0, 0, 0, 8), make_ip(10, 1, 0, 1), "eth1"),
    make_route(make_cidr(10, 1, 0, 0, 16), 0, "eth1"),
    make_route(make_cidr(10, 1, 2, 0, 24), 0, "eth2"),
    make_route(make_cidr(10, 1, 2, 128, 25), make_ip(10, 1, 2, 1), "eth2"),
    make_route(make_cidr(192, 168, 0, 0, 16), make_ip(10, 1, 0, 1), "eth1")
  };
  for (int i = 0; i < 7; i++) {
    table_add(table, routes[i]);
  }

  // 132.178.227.50 -> 132.178.227.0/25 dev eno1 direct
  match = table_lookup(table, make_ip(132, 178, 227, 50));
  TEST_ASSERT_NOT_NULL(match);
  TEST_ASSERT_EQUAL_UINT32(make_ip(132, 178, 227, 0), match->dest.addr);
  TEST_ASSERT_EQUAL_UINT8(25, match->dest.prefix);
  TEST_ASSERT_EQUAL_UINT32(0, match->gateway);
  TEST_ASSERT_EQUAL_STRING("eno1", match->dev);

  // 8.8.8.8 -> default (0.0.0.0/0) via 132.178.227.1 dev eno1
  match = table_lookup(table, make_ip(8, 8, 8, 8));
  TEST_ASSERT_NOT_NULL(match);
  TEST_ASSERT_EQUAL_UINT32(0, match->dest.addr);
  TEST_ASSERT_EQUAL_UINT8(0, match->dest.prefix);
  TEST_ASSERT_EQUAL_UINT32(make_ip(132, 178, 227, 1), match->gateway);
  TEST_ASSERT_EQUAL_STRING("eno1", match->dev);

  // 10.1.2.5 -> 10.1.2.0/24 dev eth2 direct
  match = table_lookup(table, make_ip(10, 1, 2, 5));
  TEST_ASSERT_NOT_NULL(match);
  TEST_ASSERT_EQUAL_UINT32(make_ip(10, 1, 2, 0), match->dest.addr);
  TEST_ASSERT_EQUAL_UINT8(24, match->dest.prefix);
  TEST_ASSERT_EQUAL_UINT32(0, match->gateway);
  TEST_ASSERT_EQUAL_STRING("eth2", match->dev);

  // 10.1.2.200 -> 10.1.2.128/25 via 10.1.2.1 dev eth2
  match = table_lookup(table, make_ip(10, 1, 2, 200));
  TEST_ASSERT_NOT_NULL(match);
  TEST_ASSERT_EQUAL_UINT32(make_ip(10, 1, 2, 128), match->dest.addr);
  TEST_ASSERT_EQUAL_UINT8(25, match->dest.prefix);
